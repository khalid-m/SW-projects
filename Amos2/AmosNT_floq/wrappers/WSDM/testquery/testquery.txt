create function SearchFoodByDescription(charstring FoodGroupCode)
        ->charstring descr
		as	
		select descr
		from   vector r,vector s,vector m, vector a, Integer i
		where  r=callwebservice("http://ws.strikeiron.com/USDAData?WSDL",
                                        "SearchFoodByDescription",a) and 
                       a={"",FoodGroupCode}and 
                       s=r[0] and m=s[i]and descr=m[1];

create function food_cc()-> <charstring ndbnumber key, charstring foodgroupcode, charstring longdescr>
as select v[0],v[2],v[1]
   from vector of vector r, integer i, vector v
   where r = callwebservice("http://ws.strikeiron.com/USDAData?WSDL",
                            "SearchFoodByDescription",{"",""}) and
         v = r[0][i];

create_mapped_type("food",{"ndbnumber"},{"ndbnumber","foodcode","longdescr"}, "food_cc");
