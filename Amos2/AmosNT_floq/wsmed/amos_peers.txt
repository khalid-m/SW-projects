
create function states()-><charstring state, charstring statename> as 
				      select state,statename
				      from  Charstring types,  real latDegrees, real lonDegrees, real latRadians, real lonRadians 
				      where GetAllStates()=< statename ,types ,state, latDegrees , lonDegrees , latRadians , lonRadians>;



create type state properties(stab vector of vector);
create state instances :s;
set stab(:s)=vectorof(GetAllStates());



//create function state_r(integer num1, integer num2)-><charstring> as select ((stab(st))[i])[2] from  integer i , state st where  i in iota(num1-1,num2-1);
create function state_r(integer num1, integer num2)-><charstring> as select x[i][2] from  vector of vector x, integer i , state st where  i in iota(num1-1,num2-1) and x=stab(st);







create function placeinfo_r(charstring city, integer start , integer finish, charstring peername)-> charstring as
					select peername
					from charstring placename, charstring statename, real plon, real plat, charstring placetypeid,charstring country,integer availablethememask,integer population,  charstring state1
					where GetPlaceList(city+", "+state1, 100,'true')=<placename, statename, country, plon, plat, availablethememask, placetypeid,population> and state1=state_r(start,finish);

create function placeinfo_count(charstring city,integer start , integer finish, charstring peername)-> charstring as
					select peername
                                        where count(placeinfo_r(city,start,finish,peername))!=-1;
