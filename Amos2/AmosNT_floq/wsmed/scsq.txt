create function placeinfo(charstring city,charstring state)-><charstring placename, charstring placetypeid,real plon,real plat> as
					select placename,placetypeid,plon,plat 
					from charstring country,integer availablethememask,integer population, charstring statename
					where GetPlaceList(city+", "+state, 100,'true')=<placename, statename, country, plon, plat, availablethememask, placetypeid,population>;




create function cityinfo(charstring city,charstring state)-> <real latD , real lonD> as
				   select latD, lonD
				   from charstring resultState,charstring types, charstring placename  , real latRadians ,real  lonRadians
                                   where GetPlaceDetails(city,state)=< placename ,resultState, types , latD , lonD, latRadians , lonRadians>;



create function stateinfo()-><charstring state, charstring statename> as 
				      select statename,state 
				      from  Charstring types,  real latDegrees, real lonDegrees, real latRadians, real lonRadians 
				      where GetAllStates()=< statename ,types ,state, latDegrees , lonDegrees , latRadians , lonRadians>;



create function state_range(integer num1, integer num2)-><charstring> as
select x[i][1] from vector of vector x, integer i where x = vectorof(states()) and i in iota(num1,num2);

create function state_stream(integer num1, integer num2)-> stream as
select streamof(state_range(num1,num2));
